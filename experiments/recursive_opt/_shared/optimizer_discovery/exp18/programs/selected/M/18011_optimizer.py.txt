import random
import math

def propose(history, bounds, seed):
    """Optimize a smooth black-box function using initial axis sampling and a diagonal quadratic model."""
    rng = random.Random(seed + len(history))
    dim = len(bounds)
    n = len(history)
    lows = [lo for lo, hi in bounds]
    highs = [hi for lo, hi in bounds]
    ranges = [hi - lo for lo, hi in bounds]

    def clamp(x):
        return [max(lows[i], min(highs[i], x[i])) for i in range(dim)]

    def is_dup(x, tol=1e-9):
        for row in history:
            if all(abs(x[i] - row["x"][i]) < tol for i in range(dim)):
                return True
        return False

    # First point: center of the box
    if n == 0:
        return [(lo + hi) / 2.0 for lo, hi in bounds]

    # Initial axis sampling: center ± delta_i for each dimension
    if n <= 2 * dim:
        # center is the first evaluated point
        center = history[0]["x"]
        idx = (n - 1) // 2
        sign = 1 if (n - 1) % 2 == 0 else -1
        delta = 0.2 * ranges[idx]
        x = center[:]
        x[idx] += sign * delta
        return clamp(x)

    # ---- Model-based phase ----
    best_row = min(history, key=lambda row: row["value"])
    best_x = best_row["x"]

    # Fit diagonal quadratic: f(x) ≈ c + Σ a_i (x_i - b_i) + Σ b_i (x_i - b_i)^2
    p = 1 + 2 * dim
    X = [[0.0] * p for _ in range(n)]
    y = [0.0] * n
    for r, row in enumerate(history):
        x = row["x"]
        y[r] = row["value"]
        X[r][0] = 1.0
        for i in range(dim):
            d = x[i] - best_x[i]
            X[r][1 + i] = d
            X[r][1 + dim + i] = d * d

    # Normal equations with ridge
    A = [[0.0] * p for _ in range(p)]
    bvec = [0.0] * p
    for r in range(n):
        for i in range(p):
            bvec[i] += X[r][i] * y[r]
            for j in range(p):
                A[i][j] += X[r][i] * X[r][j]
    lam = 1e-6
    for i in range(p):
        A[i][i] += lam

    # Solve linear system via Gaussian elimination
    M = [A[i][:] + [bvec[i]] for i in range(p)]
    for col in range(p):
        pivot = None
        for r in range(col, p):
            if abs(M[r][col]) > 1e-12:
                pivot = r
                break
        if pivot is None:
            # Fallback: random perturbation of best
            scale = 0.1 * (1.0 - n / 40.0)
            cand = [best_x[i] + rng.gauss(0, scale * ranges[i]) for i in range(dim)]
            return clamp(cand)
        if pivot != col:
            M[col], M[pivot] = M[pivot], M[col]
        pv = M[col][col]
        for c in range(col, p + 1):
            M[col][c] /= pv
        for r in range(p):
            if r != col and abs(M[r][col]) > 1e-12:
                factor = M[r][col]
                for c in range(col, p + 1):
                    M[r][c] -= factor * M[col][c]
    theta = [M[i][p] for i in range(p)]

    # Build candidate from the quadratic model
    cand = best_x[:]
    for i in range(dim):
        a_i = theta[1 + i]
        b_i = theta[1 + dim + i]
        if abs(b_i) < 1e-12:
            if a_i > 0:
                cand[i] = lows[i]
            elif a_i < 0:
                cand[i] = highs[i]
            else:
                cand[i] = best_x[i]
        elif b_i > 0:
            cand[i] = best_x[i] - a_i / (2.0 * b_i)
        else:
            # Concave: choose boundary with lower predicted value
            val_lo = theta[0] + a_i * (lows[i] - best_x[i]) + b_i * (lows[i] - best_x[i]) ** 2
            val_hi = theta[0] + a_i * (highs[i] - best_x[i]) + b_i * (highs[i] - best_x[i]) ** 2
            cand[i] = lows[i] if val_lo < val_hi else highs[i]

    cand = clamp(cand)

    # Trust region: stay near explored area
    mins = [min(row["x"][i] for row in history) for i in range(dim)]
    maxs = [max(row["x"][i] for row in history) for i in range(dim)]
    expand = [0.2 * ranges[i] for i in range(dim)]
    for i in range(dim):
        lo = max(lows[i], mins[i] - expand[i])
        hi = min(highs[i], maxs[i] + expand[i])
        cand[i] = max(lo, min(hi, cand[i]))

    # Avoid duplicates
    if is_dup(cand):
        cand = [best_x[i] + rng.gauss(0, 0.05 * ranges[i]) for i in range(dim)]
        cand = clamp(cand)
        if is_dup(cand):
            cand = [rng.uniform(lows[i], highs[i]) for i in range(dim)]

    # Occasional random exploration
    if rng.random() < 0.1:
        cand = [rng.uniform(lows[i], highs[i]) for i in range(dim)]

    return cand
